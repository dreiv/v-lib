barrel (index.ts, abbreviated), and a representative sample of icon files.

wrapper around static lucide SVG path data (see LICENSE-lucide.md).

FILE: packages/icons/src/svg.ts

import { h, type VNode } from 'vue'

export function svg(children: VNode[]) {
  return h(
    'svg',
    {
      xmlns: 'http://www.w3.org/2000/svg',
      viewBox: '0 0 24 24',
      width: '1em',
      height: '1em',
      fill: 'none',
      stroke: 'currentColor',
      'stroke-width': 2,
      'stroke-linecap': 'round',
      'stroke-linejoin': 'round',
      'aria-hidden': 'true',
      focusable: 'false',
    },
    children,
  )
}
